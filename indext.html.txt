    async function openProduct(pid) {
        if (IS_MAINTENANCE) {
            alert(MAINTENANCE_MSG);
            return;
        }
        CURRENT_PID = pid;
        $('claimModal').classList.remove('hidden');
        $('modalLoadingState').classList.add('hidden');
        $('startClaimBtn').classList.remove('hidden');
        $('modalKeyDisplay').classList.add('hidden');
        $('modalStatusText').textContent = "Đang tải thông tin...";
        $('btnText').textContent = "Đang tải...";
        $('startClaimBtn').disabled = true;
        await refreshProductState();
    }

    function closeClaimModal() { $('claimModal').classList.add('hidden'); }

    async function refreshProductState() {
      try {
        const d = await api('/state', { product: CURRENT_PID });
        if (d.error) { alert(d.error); closeClaimModal(); return; }
        drawModalState(d);
      } catch (e) { $('modalStatusText').textContent = "Lỗi kết nối máy chủ"; }
    }
