    function fileToDataUrl(file) {
      return new Promise((res, rej) => {
        const img = new Image(), u = URL.createObjectURL(file);
        img.onload = () => {
          const sizes = [[1200, 0.9], [800, 0.8], [600, 0.7], [400, 0.6], [300, 0.5]];
          for (const [max, qual] of sizes) {
            const s = Math.min(1, max / Math.max(img.width, img.height));
            const c = document.createElement('canvas');
            c.width = Math.round(img.width * s); c.height = Math.round(img.height * s);
            const x = c.getContext('2d'); x.fillStyle = '#fff'; x.fillRect(0, 0, c.width, c.height); x.drawImage(img, 0, 0, c.width, c.height);
            const out = c.toDataURL('image/jpeg', qual);
            if (out.length <= 400000) { URL.revokeObjectURL(u); return res(out); }
          }
          URL.revokeObjectURL(u); rej(new Error('Ảnh quá lớn, không thể nén.'));
        };
        img.onerror = () => { URL.revokeObjectURL(u); rej(new Error('Định dạng ảnh không hỗ trợ.')); };
        img.src = u;
      });
    }

    document.addEventListener('DOMContentLoaded', () => {
        const q = new URLSearchParams(location.search);
        if (q.has('admin')) {
            $('customerView').classList.add('hidden');
            $('adminView').classList.remove('hidden');
            adminInit();
        } else {
            if (q.get('p') && q.get('v')) {
                api('/verify', { token: q.get('v') }).then(d => {
                    if (d.ok) { history.replaceState(null, '', '?p=' + q.get('p')); window.location.reload(); }
                    else if (d.wait) { setTimeout(() => window.location.reload(), (d.wait + 1) * 1000); }
                });
            } else if (q.get('p')) {
                 openProduct(q.get('p'));
            }
            loadHome();
        }
    });
  </script>
